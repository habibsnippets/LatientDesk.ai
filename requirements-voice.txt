# Install:
#   python3 -m venv .venv && .venv/bin/pip install -r requirements-voice.txt
#
# NOTE (PyAudio on machines with libportaudio but no dev headers):
#   PyAudio needs portaudio.h and libportaudio.so to build. If you lack sudo:
#     mkdir -p .build-include .build-lib
#     curl -sL -o .build-include/portaudio.h \
#       https://raw.githubusercontent.com/PortAudio/portaudio/master/include/portaudio.h
#     ln -sf /lib64/libportaudio.so.2 .build-lib/libportaudio.so   # adapt path via `ldconfig -p | grep portaudio`
#     CFLAGS="-I$PWD/.build-include" LDFLAGS="-L$PWD/.build-lib" .venv/bin/pip install pyaudio
#
# NOTE (Piper voice download): piper-tts's downloader can be very slow. If

#   curl -sL -o voices/en_US-amy-medium.onnx \

#   curl -sL -o voices/en_US-amy-medium.onnx.json \
